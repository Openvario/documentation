\chapter*{Release 5.1.0}

\begin{flushleft}
	\textbf{Release Date: September 6, 2026}
\end{flushleft}

\section*{New Features}

\begin{itemize}
	\item Yocto Scarthgap and whole-disk SD card images
	\item Cubieboard 2 images for CH57, CH70 and PQ70
	\item Raspberry Pi CM4 images for CH57 and PQ70, with Wi-Fi, serial ports and sound
	\item Linux 6.12
	\item XCSoar 7.45.1, including libgeotiff
	\item OpenSoar 7.43-3.23.8, with an application selector
	\item Split system menus, and a logbook entry
\end{itemize}

\section*{Fixed Bugs}
\begin{itemize}
	\item Quoted display rotation value
	\item sensord-testing package build
	\item Map update script
	\item QuickMenu slot conflicts with the XCSoar default menu
	\item Missing OpenSoar language file
	\item Splash-screen handover on Cubieboard 2
	\item Image file naming
	\item CM4 kernel overlay
\end{itemize}
